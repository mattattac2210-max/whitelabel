      note: `Likely instant write-off — ${cohort === 'covid' ? '$150k' : '$' + threshold.toLocaleString()} IAWO threshold applied`,

  // Diminishing value over fractional years
  const bookVal = capped * Math.pow(1 - DV_RATE, band.yrs);
  return {

    note: `Diminishing value on $${capped.toLocaleString()} base (${band.yrs}yr)`,
  };
}

function updateCostScorePreview() {
  if (!userAge || !userVType || !userFin) return;
  // Use actual estimated annual cost as the "score" indicator
  const costs  = estimateCosts(userVType, userFin, userAge, userPrice);
  const annual = costs.annual;
  // Tier by estimated annual cost
  const tier   = annual < 10000 ? 'low' : annual < 16000 ? 'medium' : 'high';
  const tiers  = {
    low:    { label: 'Lower cost', color: 'var(--bl)' },
    medium: { label: 'Mid-range',  color: 'var(--am)' },
    high:   { label: 'High cost',  color: 'var(--gr)' },
  };
  const cfg    = tiers[tier];
  const preview = document.getElementById('cost-score-preview');
  if (!preview) return;
  preview.style.display = 'block';
  const lbl  = document.getElementById('score-label');
  const note = document.getElementById('dep-note');
  const depV = document.getElementById('dep-val');
  const totV = document.getElementById('total-cost-val');
  if (lbl)  lbl.textContent  = costs.method === 'iawo' ? 'Instant Write-Off' : 'Diminishing Value';
  if (note) note.textContent = costs.note;
  if (depV) depV.textContent = costs.dep > 0 ? '$' + costs.dep.toLocaleString() + '/yr' : '$0 (written off)';
  if (totV) totV.textContent = '~$' + annual.toLocaleString() + '/yr';

  btn.style.opacity = ready ? '1' : '.35';

  lastResult = runAlgorithm(userKmBand, userAge, userFin, userVType, userPrice);

  }

  // Headline + reason
  const hl = document.getElementById('rec-headline');
  const rt = document.getElementById('rec-reason-text');
  if (hl) { hl.textContent = lastResult.method; hl.style.color = lastResult.color; }

  const prof = PROF_DEFAULTS[userTrade] || { bizPct: 0.70 };

  if (profNote) profNote.textContent = `Based on ${profPct}% average business use for ${tradeLabel}`;

  // Init slider from km band midpoint
  const costs = estimateCosts(userVType || 'ute-4x2', userFin || 'yes', userAge || '3to5', userPrice || '30to50');
  const bizKmFromBand = KM_MID[userKmBand] || Math.round(costs.totalKm * prof.bizPct);
  const kmMidWkly = Math.round(bizKmFromBand / 52);
  const slider = document.getElementById('km-slider');
  if (slider) { slider.value = Math.max(20, Math.min(400, kmMidWkly)); updateSlider(slider.value, true); }

  buildActionTiles(lastResult);

  // Update signup badge
  const signupBadge = document.getElementById('signup-badge');
  const signupSub   = document.getElementById('signup-sub');
  if (lastResult.method.includes('Cents')) {
    if (signupBadge) { signupBadge.className = 'path-badge pb-c1'; signupBadge.textContent = '✓ Cents-per-Km'; }

    if (signupSub)     signupSub.textContent = '12-week GPS logbook — free to run, $97 to export your certified PDF';
    showPlan('c2');
  }

  go('s-recommendation');
}

// ── SLIDER UPDATE ──────────────────────────────────────────
function updateSlider(wklyKm, updateGradient = true) {
  wklyKm = parseInt(wklyKm);
  const annKm  = wklyKm * 52;
  const costs  = estimateCosts(userVType || 'ute-4x2', userFin || 'yes', userAge || '3to5', userPrice || '30to50');
  const cents  = calcCentsPerKm(annKm);
  const log    = calcLogbook(annKm, costs);
  const diff   = log - cents;
  const diff5  = diff * 5;
  const color  = diff >= 0 ? (lastResult ? lastResult.color : 'var(--y)') : 'var(--t2)';
  const isCapped = annKm >= CENTS_CAP;

  // Weekly/annual display
  const setT = (id, val, c) => {
    const e = document.getElementById(id);
    if (e) { e.textContent = val; if (c) e.style.color = c; }
  };
  setT('slider-wkly-km', wklyKm);
  setT('slider-ann-km',  annKm.toLocaleString());
  setT('rec-cents-yr',   '$' + cents.toLocaleString(), isCapped ? 'var(--t2)' : 'var(--t1)');
  setT('rec-cents-cap-note', isCapped ? 'capped at 5,000 km' : wklyKm + ' km/wk × $0.88');
  setT('rec-log-yr',     '~$' + log.toLocaleString(), color);
  setT('rec-log-basis',  Math.round(annKm / costs.totalKm * 100) + '% of ~$' + costs.annual.toLocaleString());
  setT('rec-diff-yr',    (diff >= 0 ? '+' : '') + '$' + Math.abs(diff).toLocaleString(), color);
  setT('rec-diff-5yr',   (diff >= 0 ? '+' : '') + '$' + Math.abs(diff5).toLocaleString(), color);

  // Update slider track fill gradient
  if (updateGradient) {
    const slider = document.getElementById('km-slider');
    if (slider) {
      const pct = ((wklyKm - 20) / (400 - 20)) * 100;
      slider.style.background = `linear-gradient(90deg, var(--y) ${pct}%, rgba(255,255,255,.12) ${pct}%)`;
    }
  }

  // Rebuild action tiles if recommendation flips
  if (lastResult) {
    const newMethod = diff >= 0 ? 'Logbook Method' : 'Cents Per Kilometre';
    if (newMethod !== lastResult._sliderMethod) {
      lastResult._sliderMethod = newMethod;
      buildActionTiles({ ...lastResult, method: newMethod, diff, diff5yr: diff5 });
    }
  }
}
